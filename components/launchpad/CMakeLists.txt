cmake_minimum_required(VERSION 3.22)
project(ClavisLaunchpad VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.5 REQUIRED COMPONENTS Qml Quick Gui Network Concurrent)
find_package(LayerShellQt 6.6 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
qt_policy(SET QTP0001 NEW)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    qt_policy(SET QTP0004 NEW)
endif()
qt_add_executable(clavis-launchpad src/main.cpp src/appmodel.h src/wallpaper.h src/iconprovider.h)
qt_add_qml_module(clavis-launchpad URI Launchpad VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(clavis-launchpad PRIVATE Qt6::Qml Qt6::Quick Qt6::Gui Qt6::Network
    Qt6::Concurrent LayerShellQt::Interface PkgConfig::GIO)
install(TARGETS clavis-launchpad RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
option(BUILD_TESTING "Build regression tests" OFF)
if(BUILD_TESTING)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(tst_drag tests/tst_drag.cpp src/appmodel.h)
    target_link_libraries(tst_drag PRIVATE Qt6::Quick Qt6::Test PkgConfig::GIO)
    target_include_directories(tst_drag PRIVATE src)
    target_compile_definitions(tst_drag PRIVATE QMLPATH="${CMAKE_CURRENT_SOURCE_DIR}/qml/Main.qml")
    add_test(NAME tst_drag COMMAND tst_drag)
    set_tests_properties(tst_drag PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(tst_model tests/tst_model.cpp src/appmodel.h)
    target_link_libraries(tst_model PRIVATE Qt6::Test PkgConfig::GIO)
    target_include_directories(tst_model PRIVATE src)
    add_test(NAME tst_model COMMAND tst_model)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME launcher_ipc COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ipc.py $<TARGET_FILE:clavis-launchpad>)
endif()
